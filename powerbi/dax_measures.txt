/*
CUSTOMER SALES & RETENTION ANALYSIS — VERIFIED POWER BI DAX

Source: original Power BI semantic model exported to VPAX.
These measures reproduce the published dashboard definitions. The model uses
BigQuery-backed tables/views including customer_transactions and
customer_metrics.
*/

// CUSTOMER_METRICS TABLE

Lifetime Value =
SUM(customer_metrics[total_revenue])

Customer Count =
DISTINCTCOUNT(customer_metrics[customer_id])

Segment Revenue =
SUM(customer_metrics[total_revenue])

Total Revenue (Customer) =
SUM(customer_metrics[total_revenue])


// CUSTOMER_TRANSACTIONS TABLE

Avg Order Value =
DIVIDE(
    SUM(customer_transactions[order_value]),
    DISTINCTCOUNT(customer_transactions[order_id])
)

Total Revenue =
SUM(customer_transactions[order_value])

Total Orders =
DISTINCTCOUNT(customer_transactions[order_id])

Month Number =
MONTH(customer_transactions[order_date])

Retained Customers =
VAR CurrentCustomers =
    VALUES(customer_transactions[customer_id])
VAR CurrentMonth =
    MAX(customer_transactions[order_date])
VAR PreviousMonthStart =
    EOMONTH(CurrentMonth, -2) + 1
VAR PreviousMonthEnd =
    EOMONTH(CurrentMonth, -1)
VAR PreviousCustomers =
    CALCULATETABLE(
        VALUES(customer_transactions[customer_id]),
        FILTER(
            ALL(customer_transactions),
            customer_transactions[order_date] >= PreviousMonthStart &&
            customer_transactions[order_date] <= PreviousMonthEnd
        )
    )
RETURN
COUNTROWS(
    INTERSECT(CurrentCustomers, PreviousCustomers)
)

Total Customers =
DISTINCTCOUNT(customer_transactions[customer_id])

Churn Rate =
1 - [Retention Rate]

Repeat Customers =
CALCULATE(
    DISTINCTCOUNT(customer_transactions[customer_id]),
    FILTER(
        customer_transactions,
        customer_transactions[Months Since Cohort] > 0
    )
)

Retention Rate =
DIVIDE(
    [Repeat Customers],
    [Total Customers],
    0
)

Cohort Customers =
CALCULATE(
    DISTINCTCOUNT(customer_transactions[customer_id]),
    customer_transactions[Months Since Cohort] = 0
)

Retained Customers (Cohort) =
CALCULATE(
    DISTINCTCOUNT(customer_transactions[customer_id]),
    customer_transactions[Months Since Cohort] >= 1
)

Active Customers =
DISTINCTCOUNT(customer_transactions[customer_id])

Customer Lifetime Value =
SUM(customer_transactions[order_value])

Previous Month Customers =
VAR CurrentMonth =
    MAX(customer_transactions[order_date])
VAR PreviousMonthStart =
    EOMONTH(CurrentMonth, -2) + 1
VAR PreviousMonthEnd =
    EOMONTH(CurrentMonth, -1)
RETURN
CALCULATE(
    DISTINCTCOUNT(customer_transactions[customer_id]),
    FILTER(
        ALL(customer_transactions),
        customer_transactions[order_date] >= PreviousMonthStart &&
        customer_transactions[order_date] <= PreviousMonthEnd
    )
)

Average Retention Rate =
AVERAGEX(
    VALUES(customer_transactions[Month Name]),
    [Retention Rate]
)


// VERIFIED CALCULATED COLUMNS ON CUSTOMER_TRANSACTIONS

Cohort Month =
CALCULATE(
    MIN(customer_transactions[order_date]),
    ALLEXCEPT(customer_transactions, customer_transactions[customer_id])
)

Cohort =
FORMAT(customer_transactions[Cohort Month], "YYYY-MM")

Activity Month =
FORMAT(customer_transactions[order_date], "YYYY-MM")

Months Since Cohort =
DATEDIFF(
    customer_transactions[Cohort Month],
    customer_transactions[order_date],
    MONTH
)

Month Name =
FORMAT(customer_transactions[order_date], "MMMM")
